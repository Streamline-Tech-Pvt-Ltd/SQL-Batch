# USE JOIN:

1. INNER JOIN: 
mysql> select Company.ID,Company.Company_Name,Company.Location
    -> from Company
    -> INNER JOIN Interns
    -> on Interns.ID = Company.ID;
+----+-----------------+----------+
| ID | Company_Name    | Location |
+----+-----------------+----------+
|  1 | TCS             | Pune     |
|  1 | TCS             | Pune     |
|  2 | Infosys         | Mumbai   |
|  3 | Streamline Tech | Pune     |
+----+-----------------+----------+

# INNER JOIN returns only records that have a matching value in both tables.
 Neha has ID = NULL so she will not appear in the INNER JOIN result.


2. LEFT JOIN:
mysql> select Company.ID,Company.Company_Name,Company.Location
    -> from Company
    -> LEFT JOIN Interns
    -> on Interns.ID = Company.ID;
+----+-----------------+-----------+
| ID | Company_Name    | Location  |
+----+-----------------+-----------+
|  1 | TCS             | Pune      |
|  1 | TCS             | Pune      |
|  2 | Infosys         | Mumbai    |
|  3 | Streamline Tech | Pune      |
|  4 | Wipro           | Bangalore |
+----+-----------------+-----------+

# All records from the LEFT table + matching records from the RIGHT table.


3. RIGHT JOIN:
mysql> select C.ID,C.Company_Name,C.Location
    -> from Company AS C
    -> RIGHT JOIN Interns
    -> on Interns.ID = C.ID;
+------+-----------------+----------+
| ID   | Company_Name    | Location |
+------+-----------------+----------+
| NULL | NULL            | NULL     |
|    1 | TCS             | Pune     |
|    1 | TCS             | Pune     |
|    2 | Infosys         | Mumbai   |
|    3 | Streamline Tech | Pune     |
+------+-----------------+----------+

# All records from the RIGHT table + matching records from the LEFT table.

4. FULL OUTER JOIN:
mysql> select C.ID,C.Company_Name,C.Location
    -> from Company AS C
    -> LEFT JOIN Interns
    -> on Interns.ID = C.ID
    -> UNION
    -> select C.ID,C.Company_Name,C.Location
    -> from Company AS C
    -> RIGHT JOIN Interns
    -> on Interns.ID = C.ID;
+------+-----------------+-----------+
| ID   | Company_Name    | Location  |
+------+-----------------+-----------+
|    1 | TCS             | Pune      |
|    2 | Infosys         | Mumbai    |
|    3 | Streamline Tech | Pune      |
|    4 | Wipro           | Bangalore |
| NULL | NULL            | NULL      |
+------+-----------------+-----------+

# UNION combines both results and removes duplicate rows.
